\begin{frame}{Rev-5 in one slide}
\begin{itemize}
\item Rev-4's diagnosis: the model predicts \emph{where capital sits}, not \emph{which bins will pay} (mass-flow $\to$ return corr.\ 0.009)
\item Rev-5 asked: does concentration predict returns? \textbf{Pre-registered test $\to$ honest negative.} No return mechanism built.
\item Along the way: found and fixed \textbf{consistent $10^3$--$10^6\times$ unit errors} in share counts (SRE backfill $=$ \$5{,}000T) that corrupted every cap-weighted object in rev-4
\item Three \textbf{risk metrics} as diagnostics: ensemble forecast intervals, fragility index, portfolio risk
\item Clean-data re-calibration: distribution hindcasts improve markedly
\item \textbf{No new paper track. No real-capital deployment.}
\end{itemize}
\end{frame}

\begin{frame}{Data integrity fix}
\begin{itemize}
\item Jump-based cleaner missed \emph{consistent} wrong units: SRE backfill \$5,000T $\to$ top-lane share flipped 0.999/0.002 monthly; QCOM filed era \$136T; EOG, SYK, HAL, Citi \dots
\item Fix: segment split-adjusted shares at $>20\times$ jumps; check each segment vs.\ \textbf{absolute anchors} --- cap in [\$200M, \$4T], daily turnover in [0.02\%, 100\%]
\item Rescale by $10^{\pm3}, 10^{\pm6}$ only where \emph{both} turn sane; drop (never invent) the rest: 28 segments fixed, 17 dropped
\item Verification: binnable cap 2007 \$10.2T ($\approx$ \$13T true), 2021 \$41.3T ($\approx$ \$40T)
\end{itemize}

\begin{frame}{Concentration $\to$ return: honest negative}
Pre-registered, estimated 2001--2007, judged walk-forward 2008--2022:
\begin{itemize}
\item Form~1 (threshold-linear): $b=-0.18$, $t=-0.41$, $R^2=0.002$; WF corr $-0.045$ --- \textbf{fail}
\item Form~2 (quintiles): Q5$-$Q1 $-0.59\%$/mo ($t=-0.40$) in-sample, $-0.52\%$/mo ($t=-0.52$) walk-forward --- \textbf{statistically zero}, non-monotone
\item $\mathrm{corr}(C, r^e)$: $-0.083$ / $-0.001$. Nothing tradeable.
\end{itemize}

\item HHI: [0.110, 0.175]. The width \emph{is} the finding.

\begin{frame}{Risk metric 2: fragility index}
Top-lane share and HHI as percentiles of own history:
\begin{itemize}
\item \textbf{2022-12:} $C=0.101$ (\textbf{25th pctile} --- not crowded); HHI 52nd pctile
\item Most crowded: 2004-02 (0.570), 2010-01 (0.571), 2001-02 (0.423)
\item A state descriptor, \textbf{not} a signal --- crowding does not predict returns (\S3)

\begin{frame}{Risk metric 3: portfolio risk}
\begin{itemize}
\item Predicted vol from 24-bin covariance (2001--2022): tilt \textbf{14.4\%} vs.\ market 15.9\% vs.\ SPY 15.4\% --- the tilt's failure was return-side, not risk-side
\item $P(\text{top lane unwinds next month} \mid C\text{ quintile})$: 41.8\% / 44.0\% / 35.8\% / 53.8\% / \textbf{34.0\%} --- no monotone pattern; most-crowded reverses \emph{least}
\end{itemize}
\vspace{1em}
\textbf{Clean-data hindcasts} (realized-$M$ forcing): 2008--09 RMSE \textbf{0.0899} (was 0.1685); 2020--21 \textbf{0.0714} (was 0.1110), peak amplitude exact. The distribution mechanics survive the fix and work better.
\end{frame}

\begin{frame}{Verdict}
\begin{center}
\Large\bfseries\color{nmblue} No return mechanism. No new paper track.\\ No real-capital deployment.
\end{center}
\vspace{1em}
\begin{itemize}
\item Concentration does not predict returns (honest negative, $n=263$ months)
\item Risk metrics are \textbf{diagnostics}, not signals: ensemble intervals, fragility percentiles, predicted vol / reversal odds
\item Distribution model re-calibrated on clean data; code and data preserved for any future attempt
\item Next: return-side economics (earnings, flows) --- or admit the framework prices \emph{states}, not \emph{returns}
\end{itemize}
\end{frame}
