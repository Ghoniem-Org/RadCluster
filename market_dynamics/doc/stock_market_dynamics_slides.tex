\begin{frame}{Rev-6 in one slide}
\begin{itemize}
\item Seven pre-registered return-forecast tests. \textbf{All seven failed walk-forward.} Their sections leave the report.
\item What survived: \textbf{(1) the systematic Buffett screen} --- quality/value/low-beta, backtest 2009--2022: \textbf{CAGR 14.8\% vs.\ 13.6\% SPY, Sharpe 1.27 vs.\ 0.94}; \textbf{(2) the regime engine} --- validated census, scenario engine, fragility monitor. \emph{Not} a return forecaster.
\item The screen was not pre-registered $\to$ \textbf{paper-trading protocol}. No real capital until the paper record holds.
\end{itemize}
\end{frame}

\begin{frame}{Winner: systematic Buffett screen}
The unlevered selection leg of ``Buffett's Alpha'' (Frazzini--Kabiller--Pedersen: HML/QMJ/BAB + 1.6$\times$ float leverage --- leverage \emph{not} applied):
\begin{itemize}
\item Universe: true historical S\&P\,500 members, June 2009--2021
\item Scores (cross-sectional $z$, winsorized $\pm3$, point-in-time):
\begin{itemize}
\item \textbf{Quality:} ROE (net income / equity), ROA secondary --- SEC XBRL 10-K facts
\item \textbf{Value:} book/market, earnings yield
\item \textbf{Low beta:} trailing 252-day beta vs.\ SPY
\end{itemize}
\item Composite = mean of the three; \textbf{top 30, equal-weight, rebalanced yearly} (Buffett holds for years)
\end{itemize}
\end{frame}

\begin{frame}{Buffett screen: results}
2009-07--2022-06, 156 months:
\begin{itemize}
\item \textbf{CAGR 14.8\% vs.\ 13.6\%}; Sharpe \textbf{1.27 vs.\ 0.94}; vol 11.1\% vs.\ 14.2\%; max DD $-$19.6\% vs.\ $-$20.0\%; \$1 $\to$ \textbf{\$6.01} vs.\ \$5.26
\item FF3 alpha \textbf{+0.46\%/mo, $t=2.71$} (+5.5\% ann.); Mkt 0.64; SMB $-$0.31; \textbf{HML $\approx$ 0} --- quality neutralizes value: this is \textbf{quality + low-beta}, not deep value
\item Wins down years (2011, 2018, 2022); lags raging bulls (2020: $-$0.6\% vs.\ +18.3\%)
\end{itemize}

\begin{frame}{Buffett screen: honest limits}
\begin{itemize}
\item \textbf{Not pre-registered} --- researcher degrees of freedom; weaker evidence than the negative tests
\item No float leverage, no crisis deal-flow, no transaction costs (53\% annual turnover $\approx$ 0.1--0.3\%/yr), 11 tickers excluded, one 14-year bull-market window
\item \textbf{Paper-trading protocol:} frozen design, live tracking vs.\ SPY. Real capital only if the paper record holds --- then a small pilot.
\end{itemize}
\end{frame}

\begin{frame}{Validated: the regime engine (not a return forecaster)}
\begin{itemize}
\item 24 capital-weighted states (momentum $\times$ vol $\times$ turnover $\times$ size), true historical membership, master equation $d\mathbf{c}/dt = \mathbf{P} + \mathbf{S}\mathbf{J} - \mathbf{D}\mathbf{c}$
\item Data integrity: consistent $10^3$--$10^6\times$ share-unit errors fixed (SRE backfill = \$5{,}000T); binnable cap now \$10.2T (2007) / \$41.3T (2021)
\item \textbf{Scenario engine:} 200 scenarios, 12-mo top-lane share median \textbf{0.239}, 90\% interval \textbf{[0.106, 0.503]} --- the width \emph{is} the finding
\item \textbf{Fragility monitor:} 2022-12 top-lane share 0.101 (\textbf{25th pctile} --- not crowded)
\item Standing verdict: prices \emph{states}, not \emph{returns}. Never a deployment signal.
\end{itemize}
\end{frame}

\begin{frame}{Discarded candidates}
All pre-registered, all failed walk-forward; full test docs stay in the repo:
\begin{itemize}
\item \textbf{Rev-4 tilt:} mass-flow $\to$ return corr.\ 0.009 --- reclassification, not price pressure
\item \textbf{Concentration $\to$ returns:} $t=-0.41$, WF corr $-0.045$ --- honest negative
\item \textbf{Dispersion:} all OOS $R^2$ negative in the large-cap panel
\item \textbf{Comomentum:} wrong sign, OOS $R^2=-0.082$
\item \textbf{Short interest:} published sign reversed in the modern subsample
\item \textbf{Bubble characteristics:} pseudo-$R^2=0.0134$, key predictor unavailable
\item \textbf{HF crowding (13F proxy):} all pass criteria failed; marginal signal died under size control
\end{itemize}
\end{frame}

\begin{frame}{Verdict}
\begin{center}
\Large\bfseries\color{nmblue} One paper-trading winner. One validated diagnostic.\\ No real-capital deployment yet.
\end{center}
\vspace{1em}
\begin{itemize}
\item Buffett screen: promising backtest, frozen, paper-traded forward
\item Regime engine: census + scenarios + fragility monitor --- diagnostics, not signals
\item Everything else: tested, failed, documented, removed
\end{itemize}
\end{frame}
