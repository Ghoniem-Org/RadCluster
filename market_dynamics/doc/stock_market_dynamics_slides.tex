\begin{frame}{Objectives}
\begin{itemize}
\item Four regime axes: momentum $\times$ volatility $\times$ turnover $\times$ size $= 24$ bins
\item \textbf{Capital-weighted} bin shares (not headcounts)
\item \textbf{True historical} S\&P\,500 membership --- never backfilled
\item Zero-lookahead walk-forward forecast $\to$ monthly \textbf{tilt portfolio} with frozen weights
\item Benchmarks: SPY buy-and-hold; \textbf{equal-weight historical members}
\item Paper record $=$ \textbf{prerequisite} for any real-capital deployment
\end{itemize}

\begin{frame}{Master equation}
\begin{equation*}
\frac{d\mathbf{c}}{dt} = (T_{\mathrm{eff}}-I)\mathbf{c} + A(M)\,\mathbf{c} + H(\mathbf{c})\,\mathbf{c} + \mathbf{e}
\end{equation*}
\begin{itemize}
\item \textbf{Drift:} cap-weighted $T$, calm vs.\ stress regimes --- \emph{measured}
\item \textbf{Advection:} $M(t)=\bar r(t)-\bar r(t{-}12)$ translates momentum lanes; $\kappa=1$ \emph{mechanical}; reflecting extremes pile mass (spikes)
\item \textbf{Herding / sticky drift:} calibrated $h_1=\lambda_1=0$ on 2000--2007 --- \emph{off} in rev-4
\item \textbf{Damper:} impedes extreme-bin outflow post-spike; $p=1.2$, 6-mo memory, gate 0.5 --- \emph{assumed} (rev-3 episode tuning)
\end{itemize}
\end{frame}

\begin{frame}{Driver forecast: mechanical nowcast}
\begin{equation*}
\widehat M(t{+}1) = R_{\mathrm{SPY}}(t) - R_{\mathrm{SPY}}(t{-}12)
\end{equation*}
\begin{itemize}
\item From the momentum definition: $\mathrm{mom}_i(t{+}1)-\mathrm{mom}_i(t)\approx r_i(t)-r_i(t{-}12)$
\item Both terms known at month-end $t$ --- \textbf{zero lookahead}
\item 2008--2022: corr.\ 0.50, 70\% directional, RMSE 0.074
\item Expanding AR(1): corr.\ $-0.02$, 41\% directional --- \textbf{discarded}
\end{itemize}
\end{frame}

\emph{Mechanism:} $M$ translates the distribution; reflecting extremes accumulate it. The model spikes when $M$ spikes.
\end{frame}

\begin{frame}{Paper-trading protocol}
At each month-end $t$ (2008--2022):
\begin{enumerate}
\item Compute $\widehat M(t{+}1)$ mechanically
\item Simulate one step to $\widehat{\mathbf{c}}(t{+}1)$
\item Tilt: $\pi_b \propto \max(c_b + \gamma(\widehat c_b-c_b),0)$, $\gamma=1$ \emph{(additive)}
\item Cap-weight stocks within bins; \textbf{freeze} through $t{+}1$
\item 5\,bps one-way on turnover (timing bug found and fixed)
\end{enumerate}
Benchmarks: SPY; \textbf{true equal-weight historical members}.\\
Monthly equity \emph{and} frozen holdings snapshots saved.
\end{frame}

\begin{frame}{Why the tilt fails --- mechanism}
\begin{itemize}
\item corr(predicted $\Delta\widehat c_b$, realized bin return) $=$ \textbf{0.009}
\item Perfect-foresight variant (realized $M$): Sharpe 0.47 --- no better
\item \textbf{Advection mass flows are reclassification, not price pressure:} crashing stocks are reclassified into the loser bin --- it ``gains mass'' while its members lose money
\item The model predicts \emph{where capital sits}, not \emph{which bins will pay}
\item A return-predictive tilt needs a return model this framework lacks
\end{itemize}
\end{frame}

\begin{frame}{Verdict}
\begin{center}
\Large\bfseries\color{nmblue} The paper record does not support\\ real-capital deployment.
\end{center}
\vspace{1em}
\begin{itemize}

\item \textbf{Prerequisite not met.} Protocol, code, and snapshots remain so any future tilt is judged by the same bar.
\end{itemize}
\end{frame}
